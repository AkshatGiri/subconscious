process

flowchart TD
  A["Harness/Adapter Hook<br/>onConversationMessage or onAfterAgentRun"] --> B["engine.ingest(conversation)"]
  B --> C["Store raw messages<br/>conversation_entries"]
  B --> D["Update short-term state<br/>session_state summary + buffer"]
  B --> E["Enqueue subconscious task<br/>(sessionId, messageIds)"]

  E -->|queue poll interval| F["SubconsciousAgent.workQueue"]
  F --> G["processConversationTask"]
  G --> H["extractMemories<br/>(heuristics + optional LLM)"]
  H --> I["Create memory nodes"]
  I --> J["Persist long-term memory<br/>memories + memory_sources"]
  G --> K["associateAndResolve<br/>related_to/reinforces/contradicts/supersedes edges"]
  K --> L["Persist graph edges<br/>memory_edges"]
  G --> M["refreshSessionSummary<br/>(optional LLM summarize)"]

  N["Periodic consolidate timer"] --> O["consolidate()<br/>merge duplicates, archive superseded"]
  P["Periodic decay timer"] --> Q["runDecayCycle()<br/>strength decay, archive weak nodes"]

  R["Harness/Adapter Hook<br/>onBeforeAgentRun/onSessionStart"] --> S["engine.recall(query)"]
  S --> T["Flush pending queue first"]
  T --> U["Candidate retrieval<br/>(vector + lexical)"]
  U --> V["Graph expansion<br/>neighborhood traversal"]
  V --> W["Salience scoring<br/>relevance + recency + strength + associative"]
  W --> X["Select diverse top 7"]
  X --> Y["Reinforce recalled memories"]
  Y --> Z["Return formatted working-memory context"]
